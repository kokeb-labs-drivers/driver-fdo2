37// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
38// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
39// not match (KB-CPP20-007).
40
42template <typename Port>
43concept DeclaresGpioSet =
44 requires(Port& port) { port.GpioSet(CtrlPin::Power, GpioSignal::ACTIVE); };
45
47template <typename Port>
48concept HasGpioSet =
49 requires { ExactHook<Port, bool(CtrlPin, GpioSignal) noexcept>::Take(&Port::GpioSet); };

52template <typename Port>
53concept DeclaresDelayUs = requires(Port& port) { port.DelayUs(1_u32); };

57template <typename Port>
58concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };

61template <typename Port>
62concept DeclaresNowUs = requires(Port& port) { port.NowUs(); };
63
65template <typename Port>
66concept HasNowUs = requires { ExactHook<Port, uint32_t() noexcept>::Take(&Port::NowUs); };

69template <typename Port>
70concept DeclaresSetBaudRate = requires(Port& port) { port.SetBaudRate(1_u32); };

73template <typename Port>
74concept HasSetBaudRate =
75 requires { ExactHook<Port, bool(uint32_t) noexcept>::Take(&Port::SetBaudRate); };

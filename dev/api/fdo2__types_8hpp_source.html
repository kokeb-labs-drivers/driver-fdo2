500constexpr uint32_t AirtimeUs(uint32_t chars, uint32_t baud) noexcept {
501 if (baud == 0U) {

604constexpr bool ConfigTimesValid(const Config& config) noexcept {
605 return IsSupportedBaud(config.baud) &&
606 InRange(config.power_up_ms, kPowerUpMsMin, kPowerUpMsMax) &&
607 InRange(config.command_ms, kCommandMsMin, kCommandMsMax) &&
608 InRange(config.measurement_ms, kMeasurementMsMin, kMeasurementMsMax) &&
609 InRange(config.calibration_ms, kCalibrationMsMin, kCalibrationMsMax) &&
610 InRange(config.flash_ms, kFlashMsMin, kFlashMsMax) &&
611 InRange(config.logo_ms, kLogoMsMin, kLogoMsMax) && (config.margin_ms <= kMarginMsMax) &&
612 InRange(config.power_off_ms, kPowerOffMsMin, kPowerOffMsMax) &&
613 (config.broadcast_listen_ms <= kBroadcastListenMsMax);
614}

617constexpr bool ConfigLimitsValid(const Config& config) noexcept {
618 return (config.retries <= kRetriesMax) && (config.max_flash_writes >= 1U) &&
619 (config.max_flash_writes <= kFlashBudgetMax) && (config.min_steady_s <= kMinSteadyMax) &&
620 (config.cal_window_pct >= 1U) && (config.cal_window_pct <= kCalWindowPctMax) &&
621 InWindowI32(config.cal_window_min_mhpa, kCalWindowMinMhpaMin, kCalWindowMinMhpaMax);

626constexpr bool ConfigWindowsValid(const Config& config) noexcept {
627 const bool reference_ok = (config.pressure_reference_ubar == 0) ||
628 ((config.pressure_reference_ubar >= config.pressure_min_ubar) &&
629 (config.pressure_reference_ubar <= config.pressure_max_ubar));
630 return (config.o2_min_mhpa < config.o2_max_mhpa) && (config.o2_min_mhpa <= 0) &&
631 (config.temp_min_mc < config.temp_max_mc) &&
632 (config.pressure_min_ubar < config.pressure_max_ubar) && (config.pressure_min_ubar > 0) &&
633 (config.humidity_max_mpct > 0) && reference_ok;

709constexpr Result<int32_t> VolumePercentMilli(int32_t o2_mhpa, int32_t pressure_ubar) noexcept {
710 if (pressure_ubar <= 0) {
711 return Result<int32_t>::Failure(Error::InvalidArgument);
712 }
713 const auto num = static_cast<int64_t>(o2_mhpa) * 100'000;
714 const auto den = static_cast<int64_t>(pressure_ubar);
715 const int64_t half = den / 2;
716 const int64_t q = (num >= 0) ? ((num + half) / den) : -(((-num) + half) / den);
717 if ((q > INT32_MAX) || (q < INT32_MIN)) {
718 return Result<int32_t>::Failure(Error::OutOfRange);
719 }
720 return Result<int32_t>::Success(static_cast<int32_t>(q));
721}
